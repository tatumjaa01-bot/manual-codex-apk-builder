async function devilNativeSaveCanvas(canvas, filename){
  if(!canvas){
    alert('ไม่พบรูปสำหรับบันทึก');
    return false;
  }

  const data = canvas.toDataURL('image/png');

  try{
    if(window.Capacitor && Capacitor.Plugins && Capacitor.Plugins.SaveImagePlugin){
      await Capacitor.Plugins.SaveImagePlugin.savePng({
        image: data,
        filename: filename || ('devil-image-' + Date.now() + '.png')
      });
      alert('บันทึกรูปลงเครื่องแล้ว');
      return true;
    }
  }catch(e){
    alert('Native save error: ' + JSON.stringify(e));
    return false;
  }

  // fallback for browser
  const a = document.createElement('a');
  a.href = data;
  a.download = filename || ('devil-image-' + Date.now() + '.png');
  document.body.appendChild(a);
  a.click();
  a.remove();
  return true;
}

// ปุ่มเซฟฉุกเฉิน: ใช้ canvas ตัวแรกในหน้านั้น
if(!document.getElementById('devilSaveBtn')){
  const btn = document.createElement('button');
  btn.id = 'devilSaveBtn';
  btn.type = 'button';
  btn.textContent = '💾 เซฟรูป';
  btn.style.cssText = 'position:fixed;right:12px;bottom:12px;z-index:999999;background:#15f28c;color:#001b10;border:0;border-radius:14px;padding:13px 18px;font-weight:1000;font-size:15px;box-shadow:0 8px 24px rgba(0,0,0,.45);';
  btn.onclick = function(){
    const canvases = Array.from(document.querySelectorAll('canvas'));
    const canvas = canvases[canvases.length - 1] || canvases[0];
    devilNativeSaveCanvas(canvas, 'devil-export-' + Date.now() + '.png');
  };
  document.body.appendChild(btn);
}
</script>
